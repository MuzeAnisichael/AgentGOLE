## 3. 公共 API 现状

以下签名取自本分支源码。`Agent` 的记忆预设为 `"session"`、`"summary"`、`"long"` 及 `None`，也可传入实现 `Memory` 的对象。`output` 可接收 `Paper` 或 `list[str]` 等类型。

```python
class Agent:
    def __init__(
        self,
        model: str | Model | ModelClient | None = None,
        *,
        instructions: str | None = None,
        tools: Iterable[str | Tool] = (),
        memory: str | Memory | None = "session",
        knowledge: Knowledge | str | Path | list[Knowledge | str | Path] | None = None,
        knowledge_mode: str = "auto",
        paradigm: str | Paradigm = "auto",
        name: str | None = None,
        max_steps: int = 20,
        timeout: float | None = None,
        tool_mode: ToolMode | None = None,
        verbose: bool = False,
    ) -> None: ...
    @property
    def memory(self) -> Any: ...
    def run(self, input: str, *, session: str = "default", output: type | None = None) -> Result: ...
    async def arun(
        self, input: str, *, session: str = "default", output: type | None = None
    ) -> Result: ...
    def stream(
        self,
        input: str,
        *,
        session: str = "default",
        events: list[str] | None = None,
        output: type | None = None,
    ) -> Iterator[Event]: ...
    async def astream(
        self,
        input: str,
        *,
        session: str = "default",
        events: list[str] | None = None,
        output: type | None = None,
    ) -> AsyncIterator[Event]: ...

class Embedding:
    def __init__(
        self,
        spec: str,
        *,
        base_url: str | None = None,
        api_key: str | None = None,
        batch_size: int = 32,
        timeout: float = 60.0,
        retries: int = 2,
    ) -> None: ...
    def embed(self, texts: list[str]) -> list[list[float]]: ...
    async def aembed(self, texts: list[str]) -> list[list[float]]: ...

class Knowledge:
    def __init__(
        self,
        sources: str | Path | list[str | Path],
        *,
        embedding: str | Embedding | EmbeddingClient | None = None,
        store: str | Path = "./.gole/kb",
        name: str | None = None,
        description: str | None = None,
        chunk_size: int = 512,
        chunk_overlap: int = 64,
        retriever: str = "hybrid",
        rerank: str | None = None,
        backend: str = "builtin",
    ) -> None: ...
    async def abuild(self) -> dict[str, int]: ...
    def add(self, source: str | Path) -> None: ...
    async def aadd(self, source: str | Path) -> None: ...
    def search(self, query: str, k: int = 5) -> list[Chunk]: ...
    async def asearch(self, query: str, k: int = 5) -> list[Chunk]: ...
    def ask(self, question: str, *, model: str | Model | ModelClient | None = None) -> Result: ...

class Summary:
    def __init__(
        self, max_tokens: int = 2000, keep_turns: int = 4, model: str | ModelClient | None = None
    ) -> None: ...

class LongTerm:
    def __init__(
        self,
        path: str | Path = "./.gole/memory.db",
        embedding: str | Embedding | EmbeddingClient | None = None,
        top_k: int = 5,
        write: Literal["after_run", "tool"] = "after_run",
        backend: str = "builtin",
    ) -> None: ...
    def add(self, text: str, *, session: str = "default") -> None: ...
    def search(self, query: str, *, session: str = "default", k: int = 5) -> list[str]: ...
    def tools(self, session: str = "default") -> list[Tool]: ...
    def export(self, *, session: str = "default") -> dict[str, Any]: ...
    def clear(self, *, session: str = "default") -> None: ...

class Layered:
    def __init__(self, short: Window | Summary, long: LongTerm) -> None: ...

class FakeEmbedding:
    def __init__(self, dim: int = 16): ...
    async def embed(self, texts: list[str]) -> list[list[float]]: ...
```

`LongTerm` 和 `Layered` 均提供上列会话管理方法。`Window` 与 `Summary` 也提供相同签名的 `export`、`clear`。`Window` 是默认短期窗口，`FakeEmbedding` 位于 `agentgole.testing`。

- `Document` 包含 `text: str`、`source: str`、`page: int | None = None` 和独立的 `metadata: dict[str, Any]`。`Chunk` 包含 `id: str`、正文、来源、可选页码及 `score: float = 0.0`。
- `Citation` 包含 `source`、`page`、`text`，摘录最多 300 字符。`Result.citations: list[Citation]` 默认为空列表，按块 ID 去重。`Result.output: Any = None` 保存校验后的值。`retrieve` 事件携带 `query: str` 和 `chunks: list[Chunk]`。
- `EmbeddingClient.name: str`，异步 `embed(self, texts: list[str]) -> list[list[float]]`。`Retriever.name: str`、`description: str`，异步 `search(self, query: str, k: int = 5) -> list[Chunk]`。可选 `BindableMemory.bind(self, model: ModelClient, embedding: EmbeddingClient | None) -> None`。`ManagedMemory` 表示 add/search/export/clear 四个管理方法。
- `OutputValidationError(AgentGoleError)` 表示最终回答校验耗尽重试。命令行新增 `gole kb build <source...> [--embedding SPEC] [--store DIR] [--name NAME]`。

## 4. 偏差与决策

| 决策 | 原因 | 影响范围 | 需要审核方确认吗 |
| --- | --- | --- | --- |
| 结构化输出等待校验通过后发出 token。普通 ReAct 在明确的最终回答标记后即时发送 | 保证拼接的 token 等于 `Result.text`，并避免把工具请求和失败的结构化草稿展示给用户 | stream / astream | 否 |
| prompt 模式指定 output 时用显式 `Action:` 区分工具请求 | 合法 JSON 可以包含 `tool` 字段和 `Final Answer:` 字符串 | 自定义输出类型与工具并用 | 否 |
| 无显式维度的远程嵌入服务在下一次实际获取向量时检测同名模型换维 | 增量构建要求零变化时零嵌入请求。获知新维度后整库重建并记录 warning | 远程服务原地更换模型 | 否 |
| 本地文件哈希取原始字节，URL 取提取出的文本 | HTML 脚本和样式过滤后没有可检索正文变化时可复用旧块 | 远程来源的增量更新 | 否 |
| `doctor` 对 `models.embed` 实际请求一次嵌入。聊天模型沿用 D-105 | 嵌入专用模型通过 embeddings 端点验证 | CLI 能力报告 | 否 |
